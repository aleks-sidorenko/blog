### Task 2: Create `flake.nix`

**Files:**
- Create: `~/Projects/Self/blog/flake.nix`
- Create: `~/Projects/Self/blog/.gitignore`

**Step 1: Create `.gitignore`**

```
/result
/public
/.direnv
```

**Step 2: Create `flake.nix`**

```nix
{
  description = "Personal blog built with Zola";

  inputs = {
    nixpkgs.url = "github:nixos/nixpkgs/nixos-unstable";
  };

  outputs =
    { nixpkgs, ... }:
    let
      forAllSystems =
        f:
        nixpkgs.lib.genAttrs
          [
            "x86_64-linux"
            "aarch64-linux"
            "x86_64-darwin"
            "aarch64-darwin"
          ]
          (
            system:
            f {
              pkgs = nixpkgs.legacyPackages.${system};
            }
          );

      theme = {
        owner = "getzola";
        repo = "even";
        rev = "994bfdb426a75f6615a2da649d9eb57870b1ca88";
        # hash will be filled in after first build attempt
        hash = "";
      };
    in
    {
      packages = forAllSystems (
        { pkgs }:
        let
          evenTheme = pkgs.fetchFromGitHub {
            inherit (theme) owner repo rev hash;
          };
        in
        {
          default = pkgs.stdenv.mkDerivation {
            name = "blog";
            src = ./.;

            nativeBuildInputs = [ pkgs.zola ];

            buildPhase = ''
              mkdir -p themes/even
              cp -r ${evenTheme}/* themes/even/
              zola build -o $out
            '';

            dontInstall = true;
          };

          serve = pkgs.writeShellScriptBin "blog-serve" ''
            SITE_DIR="$(cd "$(dirname "$0")/../.." && pwd)"
            if [ -f "$SITE_DIR/config.toml" ]; then
              cd "$SITE_DIR"
            else
              cd "${./.}"
            fi
            TMPDIR=$(mktemp -d)
            cp -r . "$TMPDIR/"
            mkdir -p "$TMPDIR/themes/even"
            cp -r ${evenTheme}/* "$TMPDIR/themes/even/"
            chmod -R u+w "$TMPDIR"
            cd "$TMPDIR"
            ${pkgs.zola}/bin/zola serve
          '';
        }
      );

      devShells = forAllSystems (
        { pkgs }:
        {
          default = pkgs.mkShell {
            packages = [ pkgs.zola ];
          };
        }
      );
    };
}
```

**Step 3: Build to get the correct theme hash**

```bash

# This will fail with a hash mismatch. Copy the "got:" hash and update flake.nix
```

**Step 4: Update `theme.hash` in `flake.nix` with the correct hash from the error output**

**Step 5: Verify the build**

### Task 3: Create GitHub Actions deployment workflow

**Files:**
- Create: `~/Projects/Self/blog/.github/workflows/deploy.yml`

**Step 1: Create workflow directory**

```bash
mkdir -p .github/workflows
```

**Step 2: Create `deploy.yml`**

```yaml
name: Deploy to GitHub Pages

on:
  push:
    branches: [main]

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: cachix/install-nix-action@v30
        with:
          nix_path: nixpkgs=channel:nixos-unstable

      - run: nix build

      - uses: actions/upload-pages-artifact@v3
        with:
          path: result

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
```

**Step 3: Commit**

```bash
git add -A && git commit -m "ci: add GitHub Actions workflow for Pages deployment"
```
